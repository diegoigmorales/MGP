\begin{problem}
    Las diagonales de un rombo miden 24\,\mathrm{cm} y 30\,\mathrm{cm}. ¿Cuál es su área?

    \begin{enumerate}[label=\Alph*.]
        \item 180\,\mathrm{cm^2}
        \item 360\,\mathrm{cm^2}
        \item 720\,\mathrm{cm^2}
        \item 1\,440\,\mathrm{cm^2}
    \end{enumerate}
\end{problem}
