\begin{problem}
    Considera el siguiente modelo que permite estimar la estatura, en centímetros, que tendrán niñas y niños cuando terminen su desarrollo.

    \[
    \frac{\text{Estatura del padre }(\text{cm}) + \text{estatura de la madre }(\text{cm}) + 13 \cdot P}{2}
    \]

    tal que:
    \begin{itemize}
        \item $P = 1$, si su sexo es masculino.
        \item $P = -1$, si su sexo es femenino.
    \end{itemize}

    ¿Cuál será la estatura estimada de un niño de sexo masculino cuando termine su desarrollo si su madre mide 159\,\mathrm{cm} y su padre mide 168\,\mathrm{cm}?

    \begin{enumerate}[label=\Alph*.]
        \item 157\,\mathrm{cm}
        \item 163{,}5\,\mathrm{cm}
        \item 170\,\mathrm{cm}
        \item 176{,}5\,\mathrm{cm}
    \end{enumerate}
\end{problem}
