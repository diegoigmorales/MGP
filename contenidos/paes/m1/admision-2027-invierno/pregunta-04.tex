\begin{problem}
    Considera el siguiente modelo que permite estimar la estatura, en centímetros, que tendrán niñas y niños cuando terminen su desarrollo.

    \[
    \frac{\text{Estatura del padre }(\text{cm}) + \text{estatura de la madre }(\text{cm}) + 13 \cdot P}{2}
    \]

    tal que:
    \begin{itemize}
        \item $P = 1$, si su sexo es masculino.
        \item $P = -1$, si su sexo es femenino.
    \end{itemize}

    ¿Cuál será la estatura estimada de un niño de sexo masculino cuando termine su desarrollo si su madre mide \SI{159}{cm} y su padre mide \SI{168}{cm}?

    \begin{enumerate}[label=\Alph*.]
        \item \SI{157}{cm}
        \item \SI{163.5}{cm}
        \item \SI{170}{cm}
        \item \SI{176.5}{cm}
    \end{enumerate}
\end{problem}
